    "# Use available molecular descriptors as features\n",
    "# (In full pipeline, Morgan fingerprints from RDKit are used)\n",
    "\n",
    "feature_cols = [c for c in ['MW', 'LogP', 'HBD', 'HBA', 'TPSA', 'RotBonds']\n",
    "                if c in df.columns]\n",
    "\n",
    "print(f'Features used: {feature_cols}')\n",
    "\n",
    "X = df[feature_cols].fillna(df[feature_cols].median())\n",
    "y = df['activity_label']\n",
    "\n",
    "# Scale features\n",
    "scaler = StandardScaler()\n",
    "X_scaled = scaler.fit_transform(X)\n",
    "X_scaled = pd.DataFrame(X_scaled, columns=feature_cols)\n",
    "\n",
    "# Split\n",
    "X_train, X_test, y_train, y_test = train_test_split(\n",
    "    X_scaled, y, test_size=0.2, random_state=42, stratify=y\n",
    ")\n",
    "\n",
    "print(f'Train: {X_train.shape[0]} | Test: {X_test.shape[0]}')\n",
    "print(f'Active in train: {y_train.sum()} | Inactive: {(y_train==0).sum()}')"

    "## 5. Model Training & Evaluation <a id='5'></a>"
   ]
  },
  {
   "cell_type": "code",
   "execution_count": null,
   "metadata": {},
   "outputs": [],
   "source": [
    "# Train multiple classifiers\n",
    "models = {\n",
    "    'Logistic Regression' : LogisticRegression(random_state=42, max_iter=1000),\n",
    "    'Random Forest'       : RandomForestClassifier(n_estimators=200, random_state=42),\n",
    "    'Gradient Boosting'   : GradientBoostingClassifier(n_estimators=200, random_state=42),\n",
    "    'SVM'                 : SVC(probability=True, random_state=42),\n",
    "}\n",
    "\n",
    "results = {}\n",
    "skf = StratifiedKFold(n_splits=5, shuffle=True, random_state=42)\n",
    "\n",
    "for name, model in models.items():\n",
    "    model.fit(X_train, y_train)\n",
    "    preds      = model.predict(X_test)\n",
    "    proba      = model.predict_proba(X_test)[:, 1]\n",
    "    cv_scores  = cross_val_score(model, X_scaled, y, cv=skf, scoring='roc_auc')\n",
    "\n",
    "    results[name] = {\n",
    "        'model'    : model,\n",
    "        'preds'    : preds,\n",
    "        'proba'    : proba,\n",
    "        'accuracy' : accuracy_score(y_test, preds),\n",
    "        'roc_auc'  : roc_auc_score(y_test, proba),\n",
    "        'mcc'      : matthews_corrcoef(y_test, preds),\n",
    "        'cv_auc'   : cv_scores.mean(),\n",
    "    }\n",
    "    print(f'✓ {name} — AUC: {results[name][\"roc_auc\"]:.4f}')\n",
    "\n",
    "# Summary\n",
    "summary = pd.DataFrame({\n",
    "    'Model'    : list(results.keys()),\n",
    "    'Accuracy' : [round(v['accuracy'], 4) for v in results.values()],\n",
    "    'ROC-AUC'  : [round(v['roc_auc'],  4) for v in results.values()],\n",
    "    'MCC'      : [round(v['mcc'],       4) for v in results.values()],\n",
    "    'CV AUC'   : [round(v['cv_auc'],    4) for v in results.values()],\n",
    "}).sort_values('ROC-AUC', ascending=False)\n",
    "\n",
    "print('\\n── Model Performance ──────────────────────────────────')\n",

    "## 6. Uncertainty Quantification <a id='6'></a>"
   ]
  },
  {
   "cell_type": "code",
   "execution_count": null,
   "metadata": {},
   "outputs": [],
   "source": [
    "# Uncertainty quantification using prediction probability spread\n",
    "best_model = results[best_name]['model']\n",
    "proba = results[best_name]['proba']\n",
    "\n",
    "# Confidence = distance from 0.5 decision boundary\n",
    "confidence = np.abs(proba - 0.5) * 2  # 0 = uncertain, 1 = certain\n",
    "\n",
    "uq_df = pd.DataFrame({\n",
    "    'predicted_prob': proba,\n",
    "    'confidence': confidence,\n",
    "    'prediction': ['Active' if p >= 0.5 else 'Inactive' for p in proba],\n",
    "    'actual': ['Active' if a == 1 else 'Inactive' for a in y_test],\n",
    "})\n",
    "uq_df['correct'] = uq_df['prediction'] == uq_df['actual']\n",
    "\n",
    "# Plot confidence vs accuracy\n",
    "bins = np.linspace(0, 1, 11)\n",
    "uq_df['confidence_bin'] = pd.cut(uq_df['confidence'], bins=bins)\n",
    "bin_acc = uq_df.groupby('confidence_bin')['correct'].mean()\n",
    "\n",
    "plt.figure(figsize=(10, 5))\n",
    "plt.bar(range(len(bin_acc)), bin_acc.values, color=ACCENT, alpha=0.85)\n",
    "plt.xticks(range(len(bin_acc)),\n",
    "           [f'{b.left:.1f}-{b.right:.1f}' for b in bin_acc.index],\n",
    "           rotation=45)\n",
    "plt.xlabel('Confidence Level', fontsize=12)\n",
    "plt.ylabel('Accuracy', fontsize=12)\n",
    "plt.title('Uncertainty Quantification — Confidence vs Accuracy',\n",
    "          fontsize=13, fontweight='bold')\n",
    "plt.axhline(y=0.5, color='white', linestyle='--', alpha=0.5)\n",
    "plt.tight_layout()\n",
    "plt.savefig('figures/uncertainty_quantification.png', dpi=150, bbox_inches='tight')\n",
    "plt.show()\n",
    "\n",
    "print(f'High confidence predictions (>0.8): {(confidence > 0.8).sum()}')\n",
    "print(f'Low confidence predictions (<0.2): {(confidence < 0.2).sum()}')"
